intelligent_control.experiments

Nine reproducible laboratories using an explicitly hypothetical tank.

Use run_unit to select a laboratory and optionally save its results. Each unitN function returns a dictionary with a JSON-compatible summary and named traces; some units add numerical extra data. Trace dictionaries hold aligned t, reference, output and control arrays. Unit 09 additionally returns diagnosis records.

The shared scalar plant uses dimensionless concentration and minutes. The separate MIMO case in Unit 04 uses sample index rather than a physical time scale. All data are synthetic, and comparisons state their measurement and model-information assumptions. Importing this module runs no experiment. Only the save functions write files; their selected output names may be replaced when a destination already exists.

 41def scenario(dt=.2, duration=80.):
 42    """Return the common reference and inlet-disturbance schedule.
 43
 44    ``dt`` and ``duration`` are in minutes and should define at least two
 45    samples for `simulate`. The time vector starts at zero. The reference is
 46    0.35 before 10 min, 0.65 until 55 min, then 0.45. The additive inlet
 47    disturbance becomes 0.08 at 45 min. Return ``(t, reference, disturbance)``
 48    as equal-length one-dimensional arrays.
 49    """
 50    t=np.arange(0,duration+dt/2,dt)
 51    r=np.where(t<10,.35,np.where(t<55,.65,.45))
 52    d=np.where(t>=45,.08,0.)
 53    return t,r,d

 56def simulate(controller, tank=None, dt=.2, duration=80., feedforward=False,
 57             seed=42, noise=0.):
 58    """Run a stateful scalar controller on the shared tank scenario.
 59
 60    Parameters
 61    ----------
 62    controller : callable
 63        Called as ``controller(reference, noisy_output, disturbance)``. With
 64        feedforward enabled it must also accept the keyword ``offset``.
 65    tank : Tank or None, optional
 66        Exact sampled plant; ``None`` selects the 100 L, 10 L/min model.
 67    dt, duration : float, optional
 68        Sample period and requested duration in minutes. Configure a stateful
 69        controller's own sample period consistently with ``dt``.
 70    feedforward : bool, optional
 71        Add ``offset=-disturbance`` before the controller's saturation logic.
 72    seed : int, optional
 73        Seed for independent Gaussian measurement-noise draws.
 74    noise : float, optional
 75        Measurement-noise scale in concentration units; zero gives no noise.
 76
 77    Returns
 78    -------
 79    dict
 80        Arrays ``t``, ``reference``, ``disturbance``, ``output``, ``control``
 81        and a ``metrics`` dictionary. Plant output starts at 0.35. Commands are
 82        clipped to ``[0, 1]``; the last stored command repeats the previous
 83        applied command and does not advance the plant. The supplied controller
 84        is mutated, so construct a fresh instance for an independent run.
 85    """
 86    tank=Tank() if tank is None else tank
 87    t,r,d=scenario(dt,duration);rng=np.random.default_rng(seed)
 88    y=np.empty(len(t));u=np.empty(len(t));y[0]=.35
 89    for k in range(len(t)-1):
 90        observed=y[k]+noise*rng.normal()
 91        # Apply feedforward before PID saturation so antiwindup sees the true command.
 92        raw=controller(r[k],observed,d[k],offset=-d[k]) if feedforward else controller(r[k],observed,d[k])
 93        u[k]=np.clip(raw,0,1)
 94        y[k+1]=tank.step(y[k],u[k],dt,d[k])
 95    u[-1]=u[-2]
 96    return dict(t=t,reference=r,disturbance=d,output=y,control=u,
 97                metrics=control_metrics(t,r,y,u))

100def unit1():
101    """Identify an affine ARX model and check an independent tank trajectory.
102
103    Training has 601 samples at 0.5 min and additive output noise. Evaluation
104    uses a new excitation, initial condition and trajectory without output
105    noise. Return analytical-response error, fitted parameters, separate
106    one-step/free-run metrics, a held-out trace and training arrays.
107    """
108    tank=Tank();dt=.5;n=601;rng=np.random.default_rng(21)
109    u=np.repeat(rng.uniform(.1,.9,61),10)[:n]
110    y=np.empty(n);y[0]=.35
111    for k in range(n-1):y[k+1]=tank.step(y[k],u[k],dt)
112    measured=y+rng.normal(0,.002,n)
113    theta=fit_arx(u,measured)
114    # New excitation and fresh trajectory; no random split of adjacent samples.
115    ut=np.repeat(np.random.default_rng(73).uniform(.05,.95,61),10)[:n]
116    yt=np.empty(n);yt[0]=.6
117    for k in range(n-1):yt[k+1]=tank.step(yt[k],ut[k],dt)
118    predicted=arx_rollout(theta,ut,yt[0])
119    one_step=theta[0]*yt[:-1]+theta[1]*ut[:-1]+theta[2]
120    tau=-dt/np.log(theta[0]);gain=theta[1]/(1-theta[0])
121    analytic=1-np.exp(-np.arange(n)*dt/tank.tau)
122    numerical=np.empty(n);numerical[0]=0
123    for k in range(n-1):numerical[k+1]=tank.step(numerical[k],1.,dt)
124    return dict(summary={'arx_coefficients':theta.tolist(),'estimated_tau_min':tau,
125        'estimated_gain':gain,'free_run':identification_metrics(yt,predicted),
126        'one_step':identification_metrics(yt[1:],one_step),
127        'analytic_max_error':float(np.max(abs(analytic-numerical))),
128        'split':'Independent excitation trajectories, seeds 21 and 73; noise in training only'},
129        traces={'held_out':{'t':np.arange(n)*dt,'reference':yt,'output':predicted,'control':ut}},
130        extra={'training_input':u,'training_output':measured})

133def unit2():
134    """Compare three IMC-inspired PI choices and a filtered PID.
135
136    Return shared-scenario traces, tracking metrics and nominal continuous
137    closed-loop poles. Pole checks concern the unsaturated nominal linear
138    model; they do not prove global stability of the sampled bounded loops.
139    """
140    traces={};summary={}
141    for lam in (2.,5.,12.):
142        kp,ki=imc_pi(1,10,lam=lam)
143        result=simulate(PID(kp,ki));traces[f'IMC lambda={lam:g}']=result
144        # For G=1/(10s+1), PI characteristic 10s²+(1+Kp)s+Ki.
145        poles=np.roots([10,1+kp,ki])
146        summary[f'lambda_{lam:g}']={'kp':kp,'ki':ki,'poles':[[p.real,p.imag] for p in poles],
147                                    **result['metrics']}
148    result=simulate(PID(2,.2,.3));traces['filtered PID']=result
149    summary['filtered_PID']=result['metrics']
150    return dict(summary=summary,traces=traces)

153def unit3():
154    """Compare feedforward, cascade and interacting two-output structures.
155
156    The feedforward pair shares an inlet-concentration disturbance. The
157    cascade pair shares a separate actuator-load disturbance and exposes its
158    additional inner-loop measurement. Two-output loops compare decentralized
159    and static-decoupled commands. Return grouped traces, RGA and per-case
160    metrics; unlike pairs are not one universal controller ranking.
161    """
162    t,r,d=scenario();dt=t[1]-t[0];traces={};summary={}
163    traces['feedback']=simulate(PID(2,.2))
164    traces['measured feedforward']=simulate(PID(2,.2),feedforward=True)
165    # Fast flow actuator: tau_a*qdot = valve + load - q. Cascade measures q.
166    for cascade in (False,True):
167        y=np.empty(len(t));q=np.empty(len(t));u=np.zeros(len(t));y[0]=q[0]=.35
168        outer=PID(2,.2,dt=dt);inner=PID(2,2,dt=dt)
169        for k in range(len(t)-1):
170            desired=outer(r[k],y[k]);u[k]=inner(desired,q[k]) if cascade else desired
171            load=-.15 if 35<=t[k]<60 else 0.
172            q[k+1]=q[k]+dt/1.5*(u[k]+load-q[k])
173            y[k+1]=Tank().step(y[k],q[k],dt)
174        u[-1]=u[-2];label='cascade' if cascade else 'single loop with actuator'
175        traces[label]=dict(t=t,reference=r,output=y,control=u)
176        summary[label]=control_metrics(t,r,y,u)
177    # Interacting two-output first-order plant. Static decoupler preserves steady gains.
178    gain=np.array([[1.,.4],[.3,1.2]]);rga=gain*np.linalg.inv(gain).T
179    a=np.diag(np.exp(-dt/np.array([10.,15.])))
180    b=(np.eye(2)-a)@gain; req=np.array([.6,.65]);x0=gain@np.array([.3,.3])
181    for decouple in (False,True):
182        y=np.zeros((len(t),2));y[0]=x0;u=np.zeros_like(y)
183        loops=[PID(1.5,.15,bias=x0[0] if decouple else .3),
184               PID(1.5,.1,bias=x0[1] if decouple else .3)]
185        for k in range(len(t)-1):
186            rr=x0 if t[k]<10 else req
187            v=np.array([loops[j](rr[j],y[k,j]) for j in range(2)])
188            u[k]=np.clip(np.linalg.solve(gain,v) if decouple else v,0,1)
189            y[k+1]=a@y[k]+b@u[k]
190        u[-1]=u[-2];label='decoupled' if decouple else 'decentralized'
191        for j in range(2):traces[f'{label} output {j+1}']=dict(t=t,reference=np.where(t<10,x0[j],req[j]),output=y[:,j],control=u[:,j])
192        summary[label]={'final_output':y[-1].tolist(),'max_actuator':float(u.max())}
193    for label in ('feedback','measured feedforward'):summary[label]=traces[label]['metrics']
194    summary['steady_gain_matrix']=gain.tolist();summary['relative_gain_array']=rga.tolist()
195    summary['comparison_scope']='Cascade pair uses an actuator-load disturbance; feedforward pair uses an inlet-concentration disturbance.'
196    return dict(summary=summary,traces=traces)

199def unit4():
200    """Run constrained scalar and two-input full-state MPC examples.
201
202    Return scalar MPC/PI traces and a separate 80-step MIMO experiment, with
203    input-movement metrics, final states and explicit solve-failure counts.
204    The scalar predictor receives the inlet disturbance; full states are
205    available. MIMO trace time is sample index, not a specified duration.
206    """
207    dt=.2;a=np.exp(-dt/10)
208    mpc=LinearMPC([[a]],[[1-a]],horizon=20,control_horizon=5,move_weight=.05,max_move=.05)
209    trace=simulate(mpc)
210    # A second, genuinely MIMO constrained prediction problem.
211    ad=np.diag([.96,.97]);bd=(np.eye(2)-ad)@np.array([[1,.25],[.2,1.]])
212    coupled=LinearMPC(ad,bd,horizon=15,control_horizon=4,move_weight=.08,
213                      max_move=.06,initial=.3,output_bounds=(0.,.95))
214    x=np.array([.375,.36]);ys=[x.copy()];us=[]
215    for _ in range(80):
216        u=coupled([.65,.55],x);x=ad@x+bd@u;ys.append(x.copy());us.append(u)
217    ys=np.array(ys);us=np.array(us)
218    traces={'constrained MPC':trace,'PI reference':simulate(PID(2,.2))}
219    for j in range(2):
220        traces[f'MIMO output {j+1}']=dict(t=np.arange(81),reference=np.full(81,[.65,.55][j]),
221                                       output=ys[:,j],control=np.r_[us[:,j],us[-1,j]])
222    return dict(summary={'siso':trace['metrics'],'solver_failures':mpc.failures,
223        'mimo_solver_failures':coupled.failures,'max_move':float(np.max(abs(np.diff(np.r_[.35,trace['control']])))),
224        'mimo_final_output':ys[-1].tolist(),'mimo_max_output':float(ys.max()),
225        'note':'Full state is measured; the additive inlet disturbance is measured for prediction. No state estimator is assumed.'},traces=traces)

247def nonlinear_step(y,u,dt=1.):
248    """Illustrative state-dependent flushing rate; not an article benchmark."""
249    return u+(y-u)*np.exp(-dt*(1+.6*y)/10)

252def unit6():
253    """Train and evaluate neural models, then perform bounded model control.
254
255    Use 650 training transitions and 250 untouched transitions from the
256    declared synthetic nonlinear process. Return MLP/RBF one-step and recursive
257    metrics, MLP out-of-domain error, a neural model-based control trace,
258    perceptron accuracy, SOM summaries and MLP training-loss samples.
259    Hyperparameters are fixed before evaluation; low one-step error is not
260    treated as proof of free-run accuracy or safe extrapolation.
261    """
262    rng=np.random.default_rng(42);x=rng.uniform(.05,.95,(900,2))
263    target=nonlinear_step(x[:,0],x[:,1]);train=x[:650];yt=target[:650]
264    validation=x[650:];yvalidation=target[650:]
265    mlp=MLP().fit(train,yt);rbf=RBF().fit(train,yt,centers=25)
266    scores={name:identification_metrics(yvalidation,model.predict(validation))
267            for name,model in [('MLP',mlp),('RBF',rbf)]}
268    t=np.arange(201.);inputs=.5+.3*np.sin(t/9)+.1*np.sin(t/2.7)
269    y=np.empty(len(t));y[0]=.3;yp={name:np.empty(len(t)) for name in ('MLP','RBF')}
270    for p in yp.values():p[0]=y[0]
271    for k in range(len(t)-1):
272        y[k+1]=nonlinear_step(y[k],inputs[k])
273        for name,model in [('MLP',mlp),('RBF',rbf)]:yp[name][k+1]=model.predict([[yp[name][k],inputs[k]]])[0]
274    traces={name+' free run':dict(t=t,reference=y,output=yp[name],control=inputs) for name in yp}
275    for name in yp:scores[name]['free_run_rmse']=identification_metrics(y,yp[name])['rmse']
276    # Model-based one-step control, all candidate actions evaluated through learned model.
277    r=np.where(t<15,.3,np.where(t<120,.7,.45));yc=np.empty(len(t));uc=np.empty(len(t));yc[0]=.3
278    candidates=np.linspace(0,1,101);previous=.3
279    for k in range(len(t)-1):
280        pred=mlp.predict(np.column_stack([np.full(len(candidates),yc[k]),candidates]))
281        objective=(pred-r[k])**2+.0002*(candidates-previous)**2
282        uc[k]=candidates[np.argmin(objective)];previous=uc[k]
283        yc[k+1]=nonlinear_step(yc[k],uc[k])
284    uc[-1]=uc[-2];traces['neural model-based control']=dict(t=t,reference=r,output=yc,control=uc)
285    labels=np.where(train[:,0]+train[:,1]>1,1,-1);w=train_perceptron(train,labels)
286    prediction=np.where(np.column_stack([validation,np.ones(len(validation))])@w>=0,1,-1)
287    _,clusters,qe=fit_som(train)
288    ood=np.array([[1.2,.2],[-.1,.8],[.5,1.3]])
289    return dict(summary={'models':scores,'training_samples':650,'validation_samples':250,
290        'perceptron_accuracy':float(np.mean(prediction==np.where(validation.sum(1)>1,1,-1))),
291        'som_quantization_error':qe,'som_occupied_nodes':int(len(np.unique(clusters))),
292        'neural_control':control_metrics(t,r,yc,uc),'out_of_domain_rmse':identification_metrics(nonlinear_step(ood[:,0],ood[:,1]),mlp.predict(ood))['rmse'],
293        'validation_protocol':'Train normalization on 650 independent state/input transitions, validate on 250 untouched transitions, then test free-run trajectory. Hyperparameters fixed before validation.'},
294        traces=traces,extra={'mlp_loss':np.asarray(mlp.loss)})

297def tuning_objective(parameters,tau=10.):
298    """Return IAE plus 0.03 times total input variation for PI tuning.
299
300    ``parameters`` supplies proportional and integral gains to `PID`, while
301    ``tau`` is the positive tank residence time in minutes. The objective uses
302    the common 80 min scenario with 0.5 min sampling and a new controller.
303    It does not save files or assess a separate validation trajectory.
304    """
305    trace=simulate(PID(*parameters,dt=.5),Tank(100,100/tau),dt=.5)
306    m=trace['metrics']
307    return m['iae']+.03*m['total_variation']

310def unit7():
311    """Compare binary/real genetic PI tuning and a small nested MPC search.
312
313    Return best parameters, fitness histories, counted evaluations and traces.
314    PI gains are also scored on a held-out 14 min residence-time plant. The
315    separate MIMO search tunes one move penalty with a deliberately small
316    budget; results do not establish convergence to a global optimum.
317    """
318    bounds=[(.2,6.),(.01,.7)]
319    results={encoding:genetic_optimize(tuning_objective,bounds,population=18,generations=16,
320                  encoding=encoding,seed=42) for encoding in ('real','binary')}
321    traces={'IMC reference':simulate(PID(2,.2))}
322    for name,result in results.items():
323        traces[name+' GA']=simulate(PID(*result['parameters']))
324        result['held_out_tau_14']=simulate(PID(*result['parameters']),Tank(140,10))['metrics']
325    # MIMO MPC penalty tuning: identical initial states/reference and fixed solve count.
326    # A small search makes solver cost explicit, not a claim of globally optimal weights.
327    def mpc_cost(parameters):
328        rho=parameters[0];ad=np.diag([.9,.92]);bd=(np.eye(2)-ad)@np.array([[1.,.3],[.2,1.]])
329        c=LinearMPC(ad,bd,horizon=6,control_horizon=2,move_weight=rho,max_move=.12,initial=.3)
330        x=np.array([.39,.36]);cost=0.;previous=np.array([.3,.3])
331        for _ in range(12):
332            u=c([.65,.6],x);x=ad@x+bd@u
333            cost+=np.sum((x-[.65,.6])**2)+.02*np.sum((u-previous)**2);previous=u
334        return float(cost+10*c.failures)
335    results['mimo_mpc_weight']=genetic_optimize(mpc_cost,[(.001,.5)],population=4,generations=2,seed=7)
336    return dict(summary=results,traces=traces,
337                extra={name+'_best_fitness':np.array(result['history']) for name,result in results.items()})

372def save_result(unit,result,outdir):
373    """Write laboratory metrics, trace CSVs and comparison figures.
374
375    Parameters
376    ----------
377    unit : int
378        Laboratory identifier from 1 through 9.
379    result : dict
380        Result produced by the matching laboratory or `run_unit`.
381    outdir : str or pathlib.Path
382        Output directory, created with parents when needed. Existing files
383        with the selected result names are replaced.
384
385    Returns
386    -------
387    pathlib.Path
388        Destination containing ``metrics.json``, per-trace CSV files, PNG/SVG
389        figures and optional extra arrays. Unit 09 delegates to
390        `intelligent_control.markov_lab.save_markov_result`. Plotting uses the
391        noninteractive Agg backend; returned metrics are not recomputed here.
392    """
393    if unit==9:
394        from .markov_lab import save_markov_result
395        return save_markov_result(result,outdir)
396    import matplotlib
397    matplotlib.use('Agg')
398    import matplotlib.pyplot as plt
399    outdir=Path(outdir);outdir.mkdir(parents=True,exist_ok=True)
400    (outdir/'metrics.json').write_text(json.dumps(result['summary'],indent=2,allow_nan=False)+'\n',encoding='utf-8')
401    plt.rcParams.update({'font.size':10,'axes.spines.top':False,'axes.spines.right':False})
402    traces=result['traces']
403    if unit==3:
404        groups=[('Inlet disturbance',[k for k in traces if k in ('feedback','measured feedforward')]),
405                ('Actuator disturbance',[k for k in traces if k in ('cascade','single loop with actuator')]),
406                ('Coupled process',[k for k in traces if 'output' in k])]
407    elif unit==4:
408        groups=[('Single-input process',[k for k in traces if not k.startswith('MIMO')]),
