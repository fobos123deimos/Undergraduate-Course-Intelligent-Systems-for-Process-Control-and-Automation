 25def siso_step(y, y_previous, u):
 26    """Eq. (30): returns y[k+1], given y[k], y[k-1], u[k]."""
 27    return y * y_previous * (y + 2.5) / (1 + y*y*y_previous*y_previous) + u

 30def mimo_step(y, u):
 31    """Eq. (31); input/output order is [channel 1, channel 2]."""
 32    y, u = _vector(y, 2, "y"), _vector(u, 2, "u")
 33    return np.array([y[0], y[0]*y[1]]) / (1 + y[1]**2) + u

 36def artificial_data(n=1000, kind="siso", signal="train", seed=2014):
 37    """Synthetic Eq. (30)/(31) records; paper training size is 100000.
 38
 39    y has n+1 rows; u[k] advances y[k] to y[k+1]. Initial states/history
 40    are zero (an explicit repository choice, not specified in the paper).
 41    """
 42    if not isinstance(n, (int, np.integer)) or n < 1:
 43        raise ValueError("n must be a positive integer")
 44    if kind not in {"siso", "mimo"} or signal not in {"train", "validation"}:
 45        raise ValueError("kind: siso/mimo; signal: train/validation")
 46    channels = 1 if kind == "siso" else 2
 47    rng = np.random.default_rng(seed)
 48    k = np.arange(n)
 49    if signal == "train":
 50        bound = 2 if kind == "siso" else 1
 51        u = rng.uniform(-bound, bound, (n, channels))
 52    else:
 53        u = np.sin(2*np.pi*k/25)[:, None]
 54        if kind == "mimo":
 55            u = np.column_stack([u[:, 0], np.cos(2*np.pi*k/25)])
 56    y = np.zeros((n+1, channels))
 57    for i in range(n):
 58        y[i+1] = (siso_step(y[i, 0], y[i-1, 0] if i else 0., u[i, 0])
 59                  if kind == "siso" else mimo_step(y[i], u[i]))
 60    return u, y

 63@dataclass(frozen=True)
 64class PendulumParameters:
 65    """Table 3 cart/pendulum parameters with explicit SI conventions.
 66
 67    ``m`` and ``M`` are pendulum and cart masses in kg; ``l`` is the
 68    characteristic length in m used by the published equation, not an
 69    independently inferred rod geometry. ``joint_friction`` multiplies angular
 70    velocity in the joint torque (N m s), ``cart_coulomb`` is force in N,
 71    ``g`` is acceleration in m/s² and ``force_limit`` is the actuator bound in N.
 72    The field is evaluated by `pendulum_rhs`; rail impacts and stiction are absent.
 73    """
 74    m: float = 0.356
 75    M: float = 4.8
 76    l: float = 0.56
 77    joint_friction: float = 0.035
 78    cart_coulomb: float = 4.9
 79    g: float = 9.81
 80    force_limit: float = 120.

 83def pendulum_rhs(t, x, u, p=PendulumParameters()):
 84    """Eq. (32); x=[theta, theta_dot, cart_position, cart_velocity], SI.
 85
 86    theta=0 is upright. Force saturation is explicit. The printed Coulomb
 87    sign law has sign(0)=0 and does not model sticking or rail impacts.
 88    """
 89    theta, omega, _, velocity = _vector(x, 4, "x")
 90    if min(p.m, p.M, p.l) <= 0:
 91        raise ValueError("m, M and l must be positive")
 92    force = np.clip(float(u), -p.force_limit, p.force_limit)
 93    friction = p.cart_coulomb*np.sign(velocity)
 94    a = 1/(p.m+p.M)
 95    s, c = np.sin(theta), np.cos(theta)
 96    ddtheta = (p.g*s - p.m*p.l*a*omega**2*c*s - a*c*(force-friction)
 97               - p.joint_friction*omega/(p.m*p.l))/(2*p.l-p.m*p.l*a*c*c)
 98    ddx = 2*a/(2-p.m*a*c*c)*(p.m*p.l*omega**2*s - .5*p.m*p.g*c*s
 99                            + force-friction + c*p.joint_friction*omega/(2*p.l))
100    return np.array([omega, ddtheta, velocity, ddx])

103def pneumatic_rhs(t, x, mass_flows, *, load_mass=3., kappa=1.4,
104                  gas_constant=8.314, temperature=293.15):
105    """Eqs. (33)-(34): x=[position,velocity,pI,pII], SI.
106
107    mass_flows=[m1,m2,m3,m4,mbp] in kg/s, with the signs of Eq. (33).
108    Valve equations are NOT supplied in the article. Kappa=1.4 is a
109    documented ideal-air closure. The printed Rg=8.314 is dimensionally
110    suspect for kg-based flow; pass 287.05 for a specific-air-gas variant.
111    Eq. (33) omits a gravitational load despite the vertical schematic;
112    no hidden gravity compensation is introduced here.
113    """
114    pos, vel, p1, p2 = _vector(x, 4, "x")
115    m1, m2, m3, m4, mbp = _vector(mass_flows, 5, "mass_flows")
116    area, volume = 2.8e-3, 5e-4
117    if abs(pos) > .125 or min(p1,p2) <= 0 or load_mass <= 0:
118        raise ValueError("pneumatic state/load outside admissible range")
119    friction = 135*vel + 21*np.sign(vel)
120    acc = (area*(p1-p2)-friction)/(load_mass+2.9)
121    dp1 = kappa*(gas_constant*temperature*(m2-m1-mbp)-area*p1*vel)/(volume+area*pos)
122    dp2 = kappa*(-gas_constant*temperature*(m4-m3-mbp)+area*p2*vel)/(volume-area*pos)
123    return np.array([vel, acc, dp1, dp2])

126def educational_valve_flows(voltage, p1, p2, conductance=1e-8):
127    """Explicit linear mass-flow closure for practice, NOT the benchmark valve.
128
129    Positive voltage fills I and empties II. Conductance has kg/(s Pa V).
130    This deliberately simple closure omits choking and compressibility
131    within the valve, and cannot validate the published pneumatic results.
132    """
133    v = float(np.clip(voltage, -10, 10))
134    supply, exhaust = 9e5, 1e5
135    if v >= 0:
136        return np.array([0, conductance*v*max(supply-p1,0), 0,
137                         conductance*v*max(p2-exhaust,0), 0])
138    return np.array([conductance*(-v)*max(p1-exhaust,0), 0,
139                     conductance*(-v)*max(supply-p2,0), 0, 0])

142def hydrostatic_rhs(t, x, u, *, pump_speed, wheel_load=0.):
143    """Literal numerical transcription of Eq. (35) and Table 5.
144
145    x=[pump_angle,motor_angle,pressure_difference,motor_speed]. The paper
146    mixes pressure/volume units and gains incompatible with its normalized
147    angle bounds. This routine is a transcription exercise, not a validated

152    u_p, u_m = _vector(u,2,"u")
153    return np.array([
154        (-a_p+241.67*u_p)/.13,
155        (-a_m+283.33*u_m)/.22,

157        ((-6.12)**2*(-23.3)**2*.98*.697*145e-4*a_m*dp
158         -.33*(-23.3)**2*omega-wheel_load*(-6.12)*(-23.3))/16512])

161def three_tank_rhs(t, x, u, *, outlet_openings=(0.,0.,0.)):
162    """Eq. (36), with optional explicitly marked drain/fault extension.
163
164    x=[h1,h2,h3] m; u=[q1,q2] m3/s; time in seconds. All drains are closed
165    for the paper's nominal equation. Area table units are corrected to m2.
166    """
167    h = _vector(x,3,"x")
168    flow = _vector(u,2,"u")
169    opening = _vector(outlet_openings,3,"outlet_openings")
170    if np.any(h < 0) or np.any(h > .62) or np.any(flow < 0) or np.any(flow > 1e-4):
171        raise ValueError("tank levels or pump flows outside Table 6 limits")
172    if np.any(opening < 0) or np.any(opening > 1):
173        raise ValueError("outlet openings must be in [0,1]")
174    def q(d):
175        return .5e-4*.48*np.sign(d)*np.sqrt(2*9.81*abs(d))
176    q13, q32 = q(h[0]-h[2]), q(h[2]-h[1])
177    drain = opening*.5e-4*.56*np.sqrt(2*9.81*h)
178    return (np.array([-q13+flow[0], q32+flow[1], q13-q32])-drain)/.0154

181def bioreactor_rhs(t, x, u, *, gamma=.4, beta=.02):
182    """Eq. (37); state=[dimensionless biomass, substrate conversion]."""
183    c1,c2 = _vector(x,2,"x")
184    if gamma <= 0 or beta <= 0 or not (0 <= c1 <= 1 and 0 <= c2 <= 1):
185        raise ValueError("bioreactor state/parameters outside physical domain")
186    if u < 0:
187        raise ValueError("feed flow cannot be negative")
188    growth = c1*(1-c2)*np.exp(c2/gamma)
189    return np.array([-c1*u+growth, -c2*u+growth*(1+beta)/(1+beta-c2)])

206def cstr_rhs(t, x, u, *, feed_concentration=5100., gamma=.1):
207    """Eqs. (39)-(43): x=[cA,cB,T,Tc], mol/m3 and degrees C; t in hours.
208
209    u=[q/VR in h^-1, cooling in MJ/h]. Multiplication by 1000 converts
210    MJ/h to kJ/h before the coolant balance. Printed temperature labels K
211    contradict Eq. (41); Celsius is consistent with its +273.15 conversion.
212    """
213    ca,cb,temp,coolant = _vector(x,4,"x")
214    dilution,cooling = _vector(u,2,"u")
215    if temp <= -273.15 or coolant <= -273.15 or min(ca,cb,feed_concentration) < 0:
216        raise ValueError("invalid CSTR state or feed")
217    if not (5 <= dilution <= 35 and -8.5 <= cooling <= 0):
218        raise ValueError("CSTR input outside Eqs. (42)-(43)")
219    k1=1.287e12*np.exp(-9758.3/(temp+273.15))
220    k2=9.043e6*np.exp(-8560/(temp+273.15))
221    heat=-3.556e-4*(k1*(ca*4.2+cb*(-11.))+k2*ca*ca*(-41.85))
222    return np.array([-k1*ca-k2*ca*ca+(feed_concentration-ca)*dilution,
223                     k1*(ca-cb)-cb*dilution,
224                     heat+30.8285*(coolant-temp)+(104.9-temp)*dilution,
225                     86.688*(temp-coolant)+gamma*cooling*1000])

228def column_a_rhs(t, state, inputs=(2.70629,3.20629,.5,.5,1.,.5,1.)):
229    """82-state Column A balances, closed using NTNU's reference parameters.
230
231    State is [x1..x41,M1..M41]; inputs=[L,V,D,B,F,zF,qF]. Time is min,
232    flows kmol/min, holdups kmol, compositions mole fractions. The nominal
233    reflux/boilup use NTNU precision (2.70629/3.20629) rather than rounded
234    Table 9 values. Eq. (47)'s V[i+1] is corrected to V[i-1], consistent
235    with Eqs. (48),(52) and the official colamod.m implementation.
236    """
237    state = _vector(state,82,"state")
238    reflux,boilup,distillate,bottoms,feed,zf,qf = _vector(inputs,7,"inputs")
239    x,m = state[:41],state[41:]
240    if np.any(m <= 0) or np.any(x < 0) or np.any(x > 1):
241        raise ValueError("Column A requires positive holdups and compositions in [0,1]")
242    if min(reflux,boilup,distillate,bottoms,feed) < 0 or not (0<=zf<=1 and 0<=qf<=1):
243        raise ValueError("invalid Column A feed or product flows")
244    vapor=1.5*x[:-1]/(1+.5*x[:-1])
245    v=np.full(40,boilup); v[20:] += (1-qf)*feed
246    liquid=np.zeros(41)
247    liquid[1:21]=3.70629+(m[1:21]-.5)/.063
248    liquid[21:40]=2.70629+(m[21:40]-.5)/.063
249    liquid[40]=reflux
250    dm=np.zeros(41); dmx=np.zeros(41)
251    for i in range(1,40):
252        dm[i]=liquid[i+1]-liquid[i]+v[i-1]-v[i]
253        dmx[i]=liquid[i+1]*x[i+1]-liquid[i]*x[i]+v[i-1]*vapor[i-1]-v[i]*vapor[i]
254    dm[20]+=feed; dmx[20]+=feed*zf
255    dm[0]=liquid[1]-v[0]-bottoms
256    dmx[0]=liquid[1]*x[1]-v[0]*vapor[0]-bottoms*x[0]
257    dm[-1]=v[-1]-reflux-distillate
258    dmx[-1]=v[-1]*vapor[-1]-(reflux+distillate)*x[-1]
259    return np.concatenate(((dmx-x*dm)/m,dm))

283class RotorMap:
284    """Bilinear interpolation of a ROSCO rotor table, rejecting extrapolation.
285
286    Parameters
287    ----------
288    path : path-like
289        ROSCO text table containing pitch in degrees, tip-speed ratios, and
290        power/thrust/torque coefficient matrices. The repository's later NREL
291        5 MW map is under ``benchmarks/data``; it is not the recovered 2014 map.
292
293    Notes
294    -----
295    Calling the instance with ``(tip_speed_ratio, pitch_radians)`` returns
296    ``(CT, CQ)``. Both coordinates must stay within the table. ``sha256`` records
297    the exact input bytes; ``cp``, ``ct`` and ``cq`` retain the coefficient grids.
298    """
299    def __init__(self, path):
300        lines=Path(path).read_text(encoding="utf-8").splitlines()
301        numeric=[np.fromstring(line,sep=" ") for line in lines if line.strip() and not line.lstrip().startswith("#")]
302        self.pitch,self.tsr=numeric[0],numeric[1]
303        rows=len(self.tsr)
304        self.cp=np.vstack(numeric[3:3+rows])
305        self.ct=np.vstack(numeric[3+rows:3+2*rows])
306        self.cq=np.vstack(numeric[3+2*rows:3+3*rows])
307        if any(z.shape!=(rows,len(self.pitch)) for z in [self.cp,self.ct,self.cq]):
308            raise ValueError("invalid rotor-map dimensions")
309        self.sha256=hashlib.sha256(Path(path).read_bytes()).hexdigest()
310
311    def __call__(self, tip_speed_ratio, pitch_radians):
312        p=float(np.rad2deg(pitch_radians)); r=float(tip_speed_ratio)
313        if not (self.pitch[0]<=p<=self.pitch[-1] and self.tsr[0]<=r<=self.tsr[-1]):
314            raise ValueError("rotor operating point outside tabulated map; extrapolation forbidden")
315        j=min(np.searchsorted(self.pitch,p,side="right")-1,len(self.pitch)-2)
316        i=min(np.searchsorted(self.tsr,r,side="right")-1,len(self.tsr)-2)
317        a=(r-self.tsr[i])/(self.tsr[i+1]-self.tsr[i]); b=(p-self.pitch[j])/(self.pitch[j+1]-self.pitch[j])
318        def interp(z):
319            return float((1-a)*((1-b)*z[i,j]+b*z[i,j+1])+a*((1-b)*z[i+1,j]+b*z[i+1,j+1]))
320        return interp(self.ct),interp(self.cq)

323def wind_rhs(t,state,inputs,aero_map):
324    """Eqs. (55)-(58); state=[q(4),qdot(4),pitch], inputs=[wind,Tg,pitch_d].
325
326    q=[tower deflection,relative blade-tip deflection,rotor angle,generator
327    angle]; aero_map(lambda,beta) returns (CT,CQ). Pitch angles are radians.
328    The bundled ROSCO map makes a documented later-map variant runnable.
329    """
330    state=_vector(state,9,"state"); wind,torque,pitch_d=_vector(inputs,3,"inputs")
331    q,vel,beta=state[:4],state[4:8],state[8]
332    veff=wind-vel[0]-vel[1]
333    if veff<=0:
334        raise ValueError("effective wind must be positive")
335    ct,cq=aero_map(vel[2]*63/veff,beta)
336    ft=.5*1.225*np.pi*63**2*ct*veff**2
337    tr=.5*1.225*np.pi*63**3*cq*veff**2
338    M,D,K=wind_matrices()
339    acc=np.linalg.solve(M,np.array([ft,ft,tr,-torque])-D@vel-K@q)
340    return np.r_[vel,acc,(pitch_d-beta)/.1]

343def rk4(rhs,x0,dt,steps):
344    """Fixed-step RK4 for short deterministic checks; no hidden clipping."""
345    if dt<=0 or steps<1:
346        raise ValueError("dt and steps must be positive")
347    x=np.asarray(x0,dtype=float).copy(); result=[x.copy()]
348    for i in range(steps):
349        t=i*dt
350        a=rhs(t,x); b=rhs(t+dt/2,x+dt*a/2)
351        c=rhs(t+dt/2,x+dt*b/2); d=rhs(t+dt,x+dt*c)
352        x=x+dt*(a+2*b+2*c+d)/6
353        if not np.all(np.isfinite(x)):
354            raise FloatingPointError(f"integration diverged at step {i+1}")
355        result.append(x.copy())
356    return np.asarray(result)

359def load_io_csv(path, input_columns, output_columns, sample_time):
360    """Read a supplied measured dataset without inventing samples or units.
361
362    CSV must have named columns, finite numbers, and at least two rows.

378class LocalARX:
379    """Eight-local-model throttle architecture from section 4.1.
380
381    Local regressors are [y[k-1],y[k-2],u[k-1],1]. Partition features are
382    [y[k-1]-y[k-2],u[k-1]], standardized on training records only. K-means
383    nearest-centroid PWA or inverse-distance TS weighting is an explicit
384    teaching fit, not the paper's unavailable fitted partition parameters.
385    """
386    def __init__(self, rules=8, mode="soft", ridge=1e-8, seed=2014):
387        if rules<1 or mode not in {"soft","crisp"} or ridge<=0:
388            raise ValueError("invalid LocalARX configuration")
389        self.rules,self.mode,self.ridge,self.seed=rules,mode,ridge,seed
390
391    def _weights(self,z):
392        dist=((z[:,None,:]-self.centers[None,:,:])**2).sum(axis=2)
393        if self.mode=="crisp":
394            w=np.zeros_like(dist); w[np.arange(len(z)),dist.argmin(axis=1)]=1
395        else:
396            w=1/np.maximum(dist,1e-12); w/=w.sum(axis=1,keepdims=True)
397        return w
398
399    def fit(self,u,y):
400        """Fit partitions and ridge-regression coefficients on training data only.
401
402        Parameters
403        ----------
404        u, y : array-like
405            Aligned finite input/output records with at least
406            ``max(20, 5*rules)`` samples. Physical units are inherited from the
407            supplied data; canonical throttle conventions are percent and degrees.
408
409        Returns
410        -------
411        LocalARX
412            This fitted instance. Feature means/scales, centers and local
413            coefficients are retained. Underdetermined crisp regions use the
414            global ridge fit. Call this method only with the training experiment.
415        """
416        u=np.asarray(u,dtype=float).reshape(-1); y=np.asarray(y,dtype=float).reshape(-1)
417        if len(u)!=len(y) or len(y)<max(20,5*self.rules) or not np.isfinite(np.r_[u,y]).all():
